Download the data

Everything above (plus full bracket tables for single and married filers) is in one JSON file: us-tax-data-2026-2027.json. Bracket arrays are [threshold, rate] pairs applied to taxable income after the standard deduction.

Example (JavaScript)

const data = require('./us-tax-data-2026-2027.json');

function federalTax(gross, year = '2026') {
  const y = data[year];
  const taxable = Math.max(0, gross - y.standard_deduction.single);
  const b = y.brackets_single;
  let tax = 0;
  for (let i = 0; i < b.length; i++) {
    const [start, rate] = b[i];
    const end = b[i + 1] ? b[i + 1][0] : Infinity;
    if (taxable > start) tax += (Math.min(taxable, end) - start) * rate;
  }
  return Math.round(tax);
}
